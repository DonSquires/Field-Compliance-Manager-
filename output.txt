function TrendChart({ data }: { data: TrendDay[] }) {
  if (!data.length) return <div style={styles.chartLoading}>No data</div>;

  const maxScans = Math.max(...data.map(d => d.scans), 1);
  const chartH = 120;
  const barW = Math.max(8, Math.floor(560 / data.length) - 4);

  return (
    <div style={styles.chartWrap}>
      <div style={styles.chartBars}>
        {data.map(day => {
          const scanH = Math.round((day.scans / maxScans) * chartH);
          const breachH = Math.round((day.breaches / maxScans) * chartH);
          const label = day.date.slice(5); // MM-DD
          return (
            <div key={day.date} style={styles.barGroup} title={`${day.date}: ${day.scans} scans, ${day.breaches} breaches`}>
              <div style={styles.barStack}>
                <div style={{ ...styles.barScan, height: scanH, width: barW }} />
                {day.breaches > 0 && (
                  <div style={{ ...styles.barBreach, height: breachH, width: barW, marginTop: -(breachH) }} />
                )}
              </div>
              <div style={styles.barLabel}>{label}</div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
